\def\bgcell#1#2#3#4{
    \fill[bg#3] (#1, -#2 - 1) rectangle +(1, 1);
    \fill[bg#4] (#1, -#2 - 1) -- (#1 + 1, -#2 - 1) -- (#1 + 1, -#2) -- cycle;
    \draw[white, line width=0.8pt] (#1, -#2 - 1) rectangle +(1, 1);
}
